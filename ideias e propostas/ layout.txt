        // Lógica do Botão Power (Clique rápido = Sleep, Segurar 3s = Popup)
        function startPowerHold() {
            powerTimer = setTimeout(() => {
                document.getElementById('power-popup').classList.remove('hidden');
                setTimeout(() => document.getElementById('power-popup').classList.remove('opacity-0'), 10);
            }, 3000);
        }

        function endPowerHold() {
            if (powerTimer) clearTimeout(powerTimer);
        }

        function handlePowerClick() {
            // Se o popup não abriu por segurar, o clique rápido suspende a tela
            setTimeout(() => {
                if (document.getElementById('power-popup').classList.contains('hidden')) {
                    triggerSleep();
                }
            }, 100);
        }

        function triggerSleep() {
            closePowerPopup();
            const screen = document.getElementById('sleep-screen');
            screen.classList.remove('pointer-events-none');
            screen.classList.remove('opacity-0');
            showToast("Tela suspensa. Pressione qualquer tecla para voltar.");
        }

        window.addEventListener('keydown', () => {
            const screen = document.getElementById('sleep-screen');
            if (!screen.classList.contains('opacity-0')) {
                screen.classList.add('opacity-0');
                screen.classList.add('pointer-events-none');
            }
        });

        function triggerShutdown() {
            closePowerPopup();
            showToast("Desligando o HandheldOS...");
            setTimeout(() => {
                document.body.innerHTML = '<div class="h-screen w-screen bg-black flex items-center justify-center text-slate-600 font-mono text-sm">Sistema Desligado.</div>';
            }, 1500);
        }

        function closePowerPopup() {
            const popup = document.getElementById('power-popup');
            popup.classList.add('opacity-0');
            setTimeout(() => popup.classList.add('hidden'), 300);
        }

        // Controle por Teclado (Simulação Gamepad L/R e Setas)
        window.addEventListener('keydown', (e) => {
            const tabs = ['music', 'home', 'library', 'store', 'sysinfo'];
            let idx = tabs.indexOf(currentTab);

            if (e.key === 'ArrowRight' || e.key === 'e') {
                if (currentTab === 'library') {
                    selectedGameIndex = (selectedGameIndex + 1) % games.length;
                    renderLibrary();
                }
            } else if (e.key === 'ArrowLeft' || e.key === 'q') {
                if (currentTab === 'library') {
                    selectedGameIndex = (selectedGameIndex - 1 + games.length) % games.length;
                    renderLibrary();
                }
            } else if (e.key === 'r' || e.key === 'PageDown') {
                idx = (idx + 1) % tabs.length;
                switchTab(tabs[idx]);
            } else if (e.key === 'l' || e.key === 'PageUp') {
                idx = (idx - 1 + tabs.length) % tabs.length;
                switchTab(tabs[idx]);
            } else if (e.key === 's' || e.key === 'Escape') {
                toggleSettingsMenu();
            }
        });

        function showToast(msg) {
            const toast = document.getElementById('toast');
            document.getElementById('toast-msg').innerText = msg;
            toast.classList.remove('opacity-0');
            setTimeout(() => toast.classList.add('opacity-0'), 2500);
        }

        // Relógio
        setInterval(() => {
            const now = new Date();
            document.getElementById('clock').innerText = `${String(now.getHours()).padStart(2,'0')}:${String(now.getMinutes()).padStart(2,'0')}`;
        }, 1000);

        window.onload = () => {
            renderLibrary();
            switchTab('library');
        };
    </script>
