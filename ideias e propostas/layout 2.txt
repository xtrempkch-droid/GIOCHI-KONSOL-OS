function showToast(msg) {
  if (State.toast.timer) clearTimeout(State.toast.timer);
  const timer = setTimeout(() => dispatch('HIDE_TOAST'), 2000);
  dispatch('SHOW_TOAST', { msg, timer });

// ═══════════════════════════════════════════════════════════════
// REDUCER (puro: state + cmd → state)
// ═══════════════════════════════════════════════════════════════
function reduce(state, cmd, payload) {
  switch (cmd) {

    // ─── Navegação ───
    case 'NAV_UP': case 'NAV_DOWN': case 'NAV_LEFT': case 'NAV_RIGHT':
      return navigate(state, cmd.replace('NAV_', ''));

    case 'CONFIRM': {
      const zone = state.focus.zone;
      const idx = state.focus.index;

      if (zone === 'game-grid') {
        return {
          ...state,
          stack: [...state.stack, { type: 'overlay', id: 'game-detail' }],
          selectedGame: state.games[idx],
          savedFocus: { ...state.savedFocus, 'game-grid': idx },
          focus: { zone: 'game-detail-actions', index: 0 },
        };
      }
      if (zone === 'music-controls') {
        showToast(['Faixa anterior', 'Play / Pause', 'Próxima faixa'][idx]);
        return state;
      }
      if (zone === 'home-action' || zone === 'store-action') {
        return switchTab(state, 'library');
      }
      if (zone === 'menu-sidebar') {
        return { ...state, menuCategory: MENU_CATEGORIES[idx] };
      }
      if (zone === 'power-actions') {
        if (idx === 0) return { ...state, stack: state.stack.slice(0, -1), sleeping: true };
        if (idx === 1) return { ...state, stack: state.stack.slice(0, -1), shutdown: true };
        if (idx === 2) return goBack(state);
      }
      if (zone === 'game-detail-actions') {
        if (idx === 0) {
          showToast(`Iniciando ${state.selectedGame.title}...`);
          return state;
        }
        return goBack(state);
      }
      return state;
    }

    case 'BACK':
      return goBack(state);

    // ─── Tabs ───
    case 'TAB_LEFT':
      if (hasOverlay(state)) return state;
      return switchTabRelative(state, -1);

    case 'TAB_RIGHT':
      if (hasOverlay(state)) return state;
      return switchTabRelative(state, 1);

    // ─── Menu ───
    case 'TOGGLE_MENU': {
      const top = currentTop(state);
      if (top.type === 'overlay') {
        if (top.id === 'menu') return goBack(state);
        return state;
      }
      return {
        ...state,
        stack: [...state.stack, { type: 'overlay', id: 'menu' }],
        focus: { zone: 'menu-sidebar', index: 0 },
      };
    }

    // ─── Power ───
    case 'POWER_SHORT':
      if (hasOverlay(state)) return state;
      return { ...state, sleeping: true };

    case 'POWER_LONG_FIRE': {
      if (hasOverlay(state)) return state;
      return {
        ...state,
        stack: [...state.stack, { type: 'overlay', id: 'power-popup' }],
        focus: { zone: 'power-actions', index: 0 },
      };
    }

    case 'WAKE':
      return { ...state, sleeping: false };

    case 'REBOOT':
      return initialState();

    // ─── Toast ───
    case 'SHOW_TOAST':
      return { ...state, toast: { msg: payload.msg, visible: true, timer: payload.timer } };
    case 'HIDE_TOAST':
      return { ...state, toast: { ...state.toast, visible: false } };

    default:
      return state;
  }
}

function navigate(state, dir) {
  const zone = state.focus.zone;
  if (!zone) return state;
  const config = FOCUS_ZONES[zone];
  if (!config) return state;

  const count = zoneCount(state, zone);
  if (count <= 0) return state;

  const cols = Math.min(config.cols, count);
  const rows = Math.ceil(count / cols);
  const idx = state.focus.index;
  const row = Math.floor(idx / cols);
  const col = idx % cols;

  let r = row, c = col;
  if (dir === 'UP')    r = Math.max(0, row - 1);
  if (dir === 'DOWN')  r = Math.min(rows - 1, row + 1);
  if (dir === 'LEFT')  c = Math.max(0, col - 1);
  if (dir === 'RIGHT') c = Math.min(cols - 1, col + 1);

  let newIdx = r * cols + c;
  if (newIdx >= count) newIdx = count - 1;

  if (newIdx === idx) return state;
  return { ...state, focus: { ...state.focus, index: newIdx } };
}

function zoneCount(state, zone) {
  if (zone === 'game-grid') return state.games.length;
  if (zone === 'music-controls') return 3;
  if (zone === 'menu-sidebar') return MENU_CATEGORIES.length;
  if (zone === 'power-actions') return 3;
  if (zone === 'game-detail-actions') return 2;
  if (zone === 'home-action' || zone === 'store-action') return 1;
  return 0;
}

function goBack(state) {
  if (state.stack.length <= 1) return state;
  const newStack = state.stack.slice(0, -1);
  const top = newStack[newStack.length - 1];
  let newFocus;
  if (top.type === 'tab') {
    const z = tabZone(top.id);
    newFocus = { zone: z, index: z ? (state.savedFocus[z] || 0) : 0 };
  } else {
    newFocus = { zone: overlayZone(top.id), index: 0 };
  }
  return { ...state, stack: newStack, focus: newFocus };
}

function switchTab(state, tabId) {
  const z = tabZone(tabId);
  return {
    ...state,
    stack: [{ type: 'tab', id: tabId }],
    focus: { zone: z, index: z ? (state.savedFocus[z] || 0) : 0 },
  };
}

function switchTabRelative(state, dir) {
  const currentId = state.stack[0].id;
  const idx = TABS.indexOf(currentId);
  const newId = TABS[(idx + dir + TABS.length) % TABS.length];
  return switchTab(state, newId);
}

// ═══════════════════════════════════════════════════════════════
// DISPATCH + RENDER LOOP
// ═══════════════════════════════════════════════════════════════
function dispatch(cmd, payload) {
  const prev = State;
  State = reduce(State, cmd, payload);
  if (State !== prev) render();
}

    e.stopPropagation();

// POWER HOLD (progress ring + long press)
// ═══════════════════════════════════════════════════════════════
let powerHoldTimer = null;
let powerHoldStart = 0;
const POWER_HOLD_MS = 1500;
const POWER_TAP_MS = 500;

function startPowerHold() {
  if (State.sleeping || State.shutdown) return;
  powerHoldStart = Date.now();
  startPowerRing();
  powerHoldTimer = setTimeout(() => {
    powerHoldTimer = null;
    dispatch('POWER_LONG_FIRE');
    stopPowerRing();
  }, POWER_HOLD_MS);
}

function endPowerHold() {
  if (!powerHoldTimer) return;
  clearTimeout(powerHoldTimer);
  powerHoldTimer = null;
  stopPowerRing();
  const elapsed = Date.now() - powerHoldStart;
  if (elapsed < POWER_TAP_MS) {
    const hasPowerPopup = State.stack.some(s => s.id === 'power-popup');
    dispatch(hasPowerPopup ? 'BACK' : 'POWER_SHORT');
  }
}

function startPowerRing() {
  const ring = document.querySelector('.power-ring-progress');
  if (!ring) return;
  const circ = 2 * Math.PI * 16;
  ring.style.transition = 'none';
  ring.style.strokeDashoffset = circ;
  void ring.getBoundingClientRect();
  ring.style.transition = `stroke-dashoffset ${POWER_HOLD_MS}ms linear`;
  ring.style.strokeDashoffset = 0;
}

function stopPowerRing() {
  const ring = document.querySelector('.power-ring-progress');
  if (!ring) return;
  const circ = 2 * Math.PI * 16;
  ring.style.transition = 'stroke-dashoffset 0.2s ease';
  ring.style.strokeDashoffset = circ;
}

function updatePowerRing() { /* opcional: hook para re-render */ }

function bindPowerButton() {
  const btn = document.getElementById('power-btn');
  if (!btn) return;
  btn.addEventListener('pointerdown', (e) => {
    e.preventDefault();
    btn.setPointerCapture(e.pointerId);
    startPowerHold();
  });
  btn.addEventListener('pointerup', (e) => {
    e.preventDefault();
    endPowerHold();
  });
  btn.addEventListener('pointercancel', endPowerHold);
  btn.addEventListener('pointerleave', () => {
    // Em touch com capture, pointerleave pode não disparar durante o hold
    if (powerHoldTimer) endPowerHold();
  });
}

// ═══════════════════════════════════════════════════════════════
// INPUT (teclado → comandos)
// ═══════════════════════════════════════════════════════════════
const KEYMAP = {
  'ArrowUp': 'NAV_UP',
  'ArrowDown': 'NAV_DOWN',
  'ArrowLeft': 'NAV_LEFT',
  'ArrowRight': 'NAV_RIGHT',
  'Enter': 'CONFIRM',
  ' ': 'CONFIRM',
  'Escape': 'BACK',
  'Tab': 'TOGGLE_MENU',
  'q': 'TAB_LEFT',
  'e': 'TAB_RIGHT',
  'p': 'POWER_SHORT',
};

window.addEventListener('keydown', (e) => {
  // Shutdown: apenas reboot
  if (State.shutdown) {
    if (e.key.toLowerCase() === 'p' || e.key === 'Enter') {
      e.preventDefault();
      dispatch('REBOOT');
    }
    return;
  }

  // Sleep: qualquer tecla acorda
  if (State.sleeping) {
    e.preventDefault();
    dispatch('WAKE');
    return;
  }

  const key = e.key.length === 1 ? e.key.toLowerCase() : e.key;
  const cmd = KEYMAP[key];
  if (cmd) {
    e.preventDefault();
    dispatch(cmd);
  }
});

// ═══════════════════════════════════════════════════════════════
// CLICK DELEGATION (mouse/touch → comandos)
// ═══════════════════════════════════════════════════════════════
document.addEventListener('click', (e) => {
  // Tab
  const tab = e.target.closest('[data-tab]');
  if (tab) { dispatch('TAB_SWITCH', null); /* fallback */ }

  // Tab click (setas Q/E equivalentes não fazem sentido aqui)
  if (tab) {
    State = switchTab(State, tab.dataset.tab);
    render();
    return;
  }

  // Game card
  const gameCard = e.target.closest('[data-game-index]');
  if (gameCard) {
    State = { ...State, focus: { zone: 'game-grid', index: parseInt(gameCard.dataset.gameIndex) } };
    dispatch('CONFIRM');
    return;
  }

  // Music buttons
  const musicBtn = e.target.closest('[data-music-btn]');
  if (musicBtn) {
    State = { ...State, focus: { zone: 'music-controls', index: parseInt(musicBtn.dataset.musicBtn) } };
    dispatch('CONFIRM');
    return;
  }

  // Menu category
  const menuCat = e.target.closest('[data-menu-cat]');
  if (menuCat) {
    const idx = MENU_CATEGORIES.indexOf(menuCat.dataset.menuCat);
    State = { ...State, focus: { zone: 'menu-sidebar', index: idx }, menuCategory: menuCat.dataset.menuCat };
    render();
    return;
  }

  // Menu close
  if (e.target.closest('[data-action="close-menu"]')) {
    dispatch('BACK');
    return;
  }

  // Menu button in header
  if (e.target.closest('[data-action="menu"]')) {
    dispatch('TOGGLE_MENU');
    return;
  }

  // Home/Store action
  if (e.target.closest('[data-action="goto-library"]')) {
    State = switchTab(State, 'library');
    render();
    return;
  }

  // Power actions
  const powerAction = e.target.closest('[data-power-action]');
  if (powerAction) {
    State = { ...State, focus: { zone: 'power-actions', index: parseInt(powerAction.dataset.powerAction) } };
    dispatch('CONFIRM');
    return;
  }

  // Game detail actions
  const detailAction = e.target.closest('[data-detail-action]');
  if (detailAction) {
    State = { ...State, focus: { zone: 'game-detail-actions', index: parseInt(detailAction.dataset.detailAction) } };
    dispatch('CONFIRM');
    return;
  }
});

// CANVAS SCALE-TO-FIT (safe area 960×720 como referência)
// ═══════════════════════════════════════════════════════════════
function fitCanvas() {
  const vw = window.innerWidth;
  const vh = window.innerHeight;
  const canvas = document.getElementById('canvas');

  // Escala baseada na SAFE AREA, não no canvas
  const scale = Math.min(vw / 960, vh / 720);
  const scaledW = 1280 * scale;
  const scaledH = 720 * scale;
  const offsetX = (vw - scaledW) / 2;
  const offsetY = (vh - scaledH) / 2;

  canvas.style.transform = `scale(${scale})`;
  canvas.style.left = offsetX + 'px';
  canvas.style.top = offsetY + 'px';
}
window.addEventListener('resize', fitCanvas);
